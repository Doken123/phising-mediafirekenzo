<script>
// GANTI INI SAMA DOMAIN VERCEL .MY.ID LU BANGSAT!
const API_URL = "https://mediafire-nine.vercel.app"; 

async function load() {
  try {
    const r = await fetch(API_URL + '/api/config');
    const d = await r.json();
    document.getElementById('email_tujuan').value = d.email_tujuan || '';
    document.getElementById('banner_url').value = d.banner_url || '';
    document.getElementById('footer_text').value = d.footer_text || '';
  } catch(e) { console.log(e); }
}

async function simpan() {
  const payload = {
    email_tujuan: document.getElementById('email_tujuan').value,
    banner_url: document.getElementById('banner_url').value,
    footer_text: document.getElementById('footer_text').value
  };
  const r = await fetch(API_URL + '/api/config', {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify(payload)
  });
  const res = await r.json();
  document.getElementById('msg').textContent = res.message || res.error || 'Gagal bangsat!';
}
load();
</script>
